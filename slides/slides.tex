\section{Recommendation System}

\begin{frame}{MERLIN: Why Two Stages?}

\begin{center}
  \small \textbf{MERLIN} = \alert{Meta-path Embedding Ranker with Learned Integration}\\[-1pt]
  \footnotesize A scalable, explainable, multi-view similar-track retrieval system.
\end{center}

\vspace{-.08cm}
\begin{center}
  \large \alert{1M tracks}
  \quad $\longrightarrow$ \quad
  \alert{$\leq 1{,}000$ candidates}
  \quad $\longrightarrow$ \quad
  \alert{Top-20}
\end{center}

\vspace{-.02cm}
\small
\begin{columns}[T,onlytextwidth]
  \begin{column}{.48\textwidth}
    \begin{block}{Stage 1: Candidate generation}
      \begin{itemize}
        \item Search four views: audio, graph, artist distance, and tags.
        \item Favor \textbf{coverage and recall} before fine-grained ordering.
        \item Preserve coverage when one view is unavailable.
      \end{itemize}
    \end{block}
  \end{column}
  \begin{column}{.48\textwidth}
    \begin{block}{Stage 2: Learned ranking}
      \begin{itemize}
        \item Score the merged pool instead of the full catalog.
        \item Fuse 13 pair features and availability masks.
        \item Learn which evidence should dominate the final order.
      \end{itemize}
    \end{block}
  \end{column}
\end{columns}

\vspace{.24cm}
\begin{center}
  \bfseries Candidate generation protects recall; the Ranker concentrates precision.
\end{center}

\end{frame}

\begin{frame}{MERLIN: System Overview}

\begin{columns}[T,onlytextwidth]
  \begin{column}{.55\textwidth}
    \textbf{Stage 1: four specialized retrievers}

    \vspace{.14cm}
    \small
    \begin{tabular}{@{}ll@{}}
      \toprule
      \textbf{Source} & \textbf{Evidence retrieved} \\
      \midrule
      C1 Audio & acoustic similarity \\
      C2 Graph & learned relational context \\
      BFS & explicit artist proximity \\
      Tag & semantic overlap \\
      \bottomrule
    \end{tabular}

    \vspace{.3cm}
    Filter, merge, and deduplicate while preserving which sources recalled each track.
  \end{column}
  \begin{column}{.40\textwidth}
    \textbf{Stage 2: R --- Learned Ranker}

    \vspace{.1cm}
    \begin{enumerate}
      \item Build the union of at most 1,000 candidates.
      \item Compute 13 pair features and availability masks.
      \item Apply one LR scorer to produce the Top-20.
    \end{enumerate}
  \end{column}
\end{columns}

\vspace{.35cm}
\begin{block}{End-to-end path}
  \centering
  Query track $\rightarrow$ specialized retrieval $\rightarrow$ canonical union
  $\rightarrow$ learned late fusion $\rightarrow$ Top-20
\end{block}

\end{frame}

\begin{frame}{MERLIN: Scaling by Access Pattern}

\small
\renewcommand{\arraystretch}{1.12}
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{1.75cm}>{\raggedright\arraybackslash}p{8.75cm}@{}}
  \toprule
  \textbf{Workload} & \textbf{Execution pattern and rationale} \\
  \midrule
  HDF5 scan & \textbf{Python multiprocessing}: parallel I/O over $\sim$267 GB of small files, with batch checkpoints for restart. \\
  \addlinespace
  Prepare + C1 & \textbf{Spark DataFrames + ML}: distributed schema checks, joins, scaling, and PCA over one million rows. \\
  \addlinespace
  C2 learning & \textbf{\texttt{mapPartitions} + Spark Word2Vec}: reuse adjacency per worker; train over $\sim$10M walks and $\sim$400M tokens. \\
  \addlinespace
  Retrieval & \textbf{FAISS \texttt{IndexFlatIP}}: exact search over compact, normalized 128D vectors on one serving machine. \\
  \bottomrule
\end{tabular}

\vspace{.28cm}
\begin{columns}[T,onlytextwidth]
  \begin{column}{.56\textwidth}
    \textbf{Spark handles the distributed work}

    ETL, feature transformation, partitioned walks, and representation learning.
  \end{column}
  \begin{column}{.40\textwidth}
    \textbf{FAISS handles online retrieval}

    Compact exact vector search; no distributed serving layer is required.
  \end{column}
\end{columns}

\end{frame}

\begin{frame}{C1: Audio to 128D Retrieval}

\begin{center}
  \includegraphics[width=.98\textwidth]{img/c1/c1_audio_pipeline.pdf}
\end{center}

\vspace{.12cm}
\begin{columns}[T,onlytextwidth]
  \column{.48\textwidth}
  \textbf{What C1 learns from}

  \small Timbre and pitch; rhythm, tempo, and duration; loudness, confidence, and temporal segment summaries.

  \column{.48\textwidth}
  \textbf{What C1 publishes}

  \small One million finite unit vectors, a frozen preprocessing/scaler/PCA bundle, and an exact FAISS index with explicit track mapping.
\end{columns}

\vspace{.30cm}
{\small\textbf{Evidence boundary.} Artist, release, year, and popularity never enter C1; each query contributes at most 250 audio candidates.}

\vfill
{\footnotesize PCA provides a fixed linear projection of the standardized acoustic view \autocite{jolliffe2016pca}.}

\end{frame}

\begin{frame}{C2: Eligibility-Aware Walks}

\begin{center}
  \includegraphics[width=.92\textwidth]{img/c2/c2_walk_policy.pdf}
\end{center}

\vspace{-.15cm}
\begin{columns}[T,onlytextwidth]
  \column{.31\textwidth}
  \textbf{\color{projectteal}1. Check}\par
  {\small Derive the paths available at the current track.}

  \column{.31\textwidth}
  \textbf{\color{projectteal}2. Sample}\par
  {\small Choose uniformly only among eligible paths.}

  \column{.31\textwidth}
  \textbf{\color{projectteal}3. Repeat}\par
  {\small Move to another track and recompute eligibility.}
\end{columns}

\vfill
{\footnotesize No fixed path order, wasted probability, or self-transition padding.}

\end{frame}

\vspace{.38cm}
\begin{columns}[c,onlytextwidth]
  \column{.31\textwidth}
  \centering
  {\fontsize{25}{27}\selectfont\bfseries\color{projectteal}10M}\par

  \column{.34\textwidth}
  \centering
  {\fontsize{25}{27}\selectfont\bfseries\color{projectblue}399.86M}\par

  \column{.31\textwidth}

\vspace{.42cm}

Walk generation batches roots with \texttt{repartition(track\_id)} and \texttt{mapPartitions}; Word2Vec trains with 20 final partitions, window 5, and 5 iterations.}

\vfill
{\footnotesize Word2Vec learns distributional vectors from local contexts \autocite{mikolov2013efficient}. Encoder export is verified; serving still filters contextless roots.}

\end{frame}

\begin{frame}{Multi-View Candidate Pool}

\begin{center}
  \includegraphics[width=.98\textwidth]{img/c3/c3_candidate_pipeline.pdf}
\end{center}

\vspace{.10cm}
\begin{columns}[T,onlytextwidth]
  \column{.48\textwidth}
  \textbf{Recall first}

  \small Four complementary views provide bounded candidate lists; short sources are never randomly padded.

  \column{.48\textwidth}
  \textbf{One serving contract}

  \small Filter, merge, and deduplicate before ranking; preserve source evidence for auditing.
\end{columns}

\vfill
{\small\textbf{Design principle:}
R cannot recover a track missing from the candidate pool.}

\end{frame}

\begin{frame}{R: Learned Ranker}

\begin{center}
  \includegraphics[width=1.0\textwidth]{img/c3/c3_pair_construction.pdf}
\end{center}

\vspace{.20cm}
\begin{columns}[T,onlytextwidth]
  \column{.31\textwidth}
  \textbf{1. Split first}

  \small Same-song variants remain in one fold.

  \column{.31\textwidth}
  \textbf{2. Sample pairs}

  \small Weak positives plus candidate-aware and random negatives.

  \column{.31\textwidth}
  \textbf{3. Fit the scorer}

  \small Thirteen pair features train a weighted Logistic Regression model \autocite{cox1958regression}.
\end{columns}

\vfill
{\small\textbf{Output:}
a learned score for ordering candidates; final serving details remain under optimization.}

\end{frame}
