\section{Year Prediction}

\subsection{Data Contract}

\begin{frame}{Year prediction starts with the data contract}

\begin{center}
\textbf{Question:} can audio statistics estimate a song's release year?

\medskip

$1{,}000{,}000$ tracks
$\longrightarrow$ valid labels
$\longrightarrow$ audio-only baseline
$\longrightarrow$ artist-safe evaluation
\end{center}

\begin{itemize}
	\item Our controlled baseline asks how much year signal exists in audio alone, following the original MSD task.~{\scriptsize\autocite{bertinmahieux2011million}}
	\item Metadata can be evaluated later as a separate extension, as proposed by the MSD authors.~{\scriptsize\autocite{bertinmahieux2011million}}
	\item Every model must see the same rows, fields, and split for a meaningful comparison.
\end{itemize}

\medskip

\begin{block}{Intuition}
A lower error is useful only when it comes from a better model, rather than a different dataset boundary.
\end{block}

\end{frame}

\begin{frame}{Step 1: define valid supervision}

\begin{columns}[T,onlytextwidth]
	\column{.47\textwidth}
	\textbf{Label contract}

	\medskip
	\begin{tabular}{l r}
		\textbf{Label state} & \textbf{Tracks} \\
		\hline
		$1922 \leq y \leq 2011$ & $515{,}576$ \\
		$y=0$ (unknown) & $484{,}424$ \\
		Other positive & $0$ \\
	\end{tabular}

	\medskip
	$y=0$ means unknown, not a historical year; it remains unlabeled. MSD reports $515{,}576$ dated tracks spanning 1922--2011.~{\scriptsize\autocite{bertinmahieux2011million}}

	\column{.49\textwidth}
	\textbf{Predictor boundary}

	\medskip
	\setlength{\leftmargini}{1em}
	\begin{itemize}
		\item Join on the unique track ID.
		\item Baseline: 109 raw acoustic predictors.
		\item IDs support grouping and audits, never prediction.
		\item Metadata is a separate ablation; year-derived features are forbidden.
	\end{itemize}
\end{columns}

\medskip
\medskip
\medskip

\textbf{Why?} Unknown years cannot train regression. Fixed inputs prevent IDs or metadata proxies from replacing acoustic evidence.

\end{frame}

\begin{frame}{Step 2: split artists, not songs}

\begin{columns}[T,onlytextwidth]
	\column{.46\textwidth}
	\begin{block}{Song-random split}
	Songs by one artist can appear in both train and test. Voice and production style become shortcuts, making test error too optimistic.~{\scriptsize\autocite{flexer2007artist}}
	\end{block}

	\column{.50\textwidth}
	\begin{block}{Artist-disjoint split}
	The MSD year task splits artists rather than songs to avoid the ``producer effect.'' Test artists are therefore unseen during training.~{\scriptsize\autocite{bertinmahieux2011million}}
	\end{block}
\end{columns}

\medskip
\medskip
\medskip

\centering
\begin{tabular}{l r r l}
	\textbf{Split} & \textbf{Artists} & \textbf{Tracks} & \textbf{Role} \\
	\hline
	Train & $22{,}867$ & $420{,}013$ & Fit all learned parameters \\
	Validation & $2{,}534$ & $46{,}127$ & Select models and settings \\
	Test & $2{,}822$ & $49{,}436$ & One frozen final evaluation \\
\end{tabular}

\medskip

The official test has $2{,}822$ artists and $49{,}436$ tracks.~{\scriptsize\autocite{msdYearTask}} We preserve it and hash all non-test artists into train or validation.

\end{frame}

\begin{frame}{Step 3: freeze a fair model comparison}

\begin{columns}[T,onlytextwidth]
	\setlength{\leftmargini}{1em}
	\column{.47\textwidth}
	\textbf{The contract freezes}
	\begin{itemize}
		\item which rows have valid labels;
		\item the primary audio feature set;
		\item artist-level split membership.
	\end{itemize}

	\column{.49\textwidth}
	\textbf{Every model shares}
	\begin{itemize}
		\item the same prediction task;
		\item the same available evidence;
		\item the same unseen-artist test.
	\end{itemize}
\end{columns}

\medskip

\begin{center}
{	
\small
\begin{tabular}{c c c c c}
	\textbf{Train} & & \textbf{Validation} & & \textbf{Test} \\
	fit preprocessing and models & $\longrightarrow$ & choose designs & $\longrightarrow$ & report once
\end{tabular}
}
\end{center}

\begin{block}{Data contract goal}
Documenting a dataset's motivation, composition, creation, and intended use improves transparency between dataset creators and consumers.~{\scriptsize\autocite{gebru2021datasheets}}
\end{block}

\end{frame}

\subsection{Feature Engineering and Contract}

\begin{frame}{Feature engineering: make raw audio learnable}

{ \small
\setlength{\leftmargini}{1em}

\begin{columns}[T,onlytextwidth]
	\footnotesize
	\column{.31\textwidth}
	\textbf{Clean semantics}
	\begin{itemize}
		\setlength{\itemsep}{1pt}
		\item Non-positive tempo becomes missing.
		\item Absent segment aggregates become missing.
		\item Fill both with train means.
	\end{itemize}

	\column{.31\textwidth}
	\textbf{Transform and encode}
	\begin{itemize}
		\setlength{\itemsep}{1pt}
		\item Clip tempo, duration, and loudness at train 1st/99th percentiles.
		\item Apply \texttt{log1p} to tempo and duration; keep loudness in dB.
		\item Encode key and meter.
	\end{itemize}

	\column{.32\textwidth}
	\textbf{Select and scale}
	\begin{itemize}
		\setlength{\itemsep}{1pt}
		\item Remove train-zero variance columns.
		\item Freeze the retained column order.
		\item Standardize the linear view with train $\mu,\sigma$.
	\end{itemize}
\end{columns}
}

\medskip
\medskip
\medskip

\begin{block}{Key idea}
The raw values are evidence. Clip bounds and encodings are project choices; normalization and feature selection follow established guidance.~{\scriptsize\autocite{guyon2003selection}}
\end{block}

\end{frame}

\begin{frame}{Feature engineering must respect the split}


\begin{center}
\small
\begin{tabular}{c c c c c}
	\textbf{Train} & $\longrightarrow$ & \textbf{Frozen transform} & $\longrightarrow$ & \textbf{Validation/test} \\
	fit all statistics & & save fitted state & & transform only
\end{tabular}
\end{center}

\medskip
\medskip

{
\small
\setlength{\leftmargini}{1em}

\begin{columns}[T,onlytextwidth]
	\column{.47\textwidth}
	\textbf{Fit on train}
	\begin{itemize}
		\setlength{\itemsep}{1pt}
		\item imputation means and clip bounds;
		\item category maps and variance filter;
		\item scaler statistics.
	\end{itemize}

	\column{.49\textwidth}
	\textbf{Reuse unchanged}
	\begin{itemize}
		\setlength{\itemsep}{1pt}
		\item validation compares designs;
		\item test reports the frozen choice;
		\item future tracks reuse the same state.
	\end{itemize}
\end{columns}
}

\medskip
\medskip

\begin{block}{Leakage boundary}
If held-out values help fit preprocessing, evaluation has already seen the answer distribution. Learn--predict separation therefore makes every fitted transform train-only.~{\scriptsize\autocite{kaufman2011leakage}}
\end{block}

\end{frame}

\begin{frame}{Feature contract: one meaning, two model views}

\begin{columns}[T,onlytextwidth]
	\small
	\setlength{\leftmargini}{1em}
	\column{.44\textwidth}
	\begin{block}{Linear view}
	\begin{itemize}
		\setlength{\itemsep}{2pt}
		\item Imputed engineered numerics.
		\item Numeric encodings for key.
		\item Fixed-order standardized vector.
		\item Ridge, PCA, RFF, and Huber.
	\end{itemize}
	\end{block}

	\column{.49\textwidth}
	\begin{block}{Tree view}
	\begin{itemize}
		\setlength{\itemsep}{2pt}
		\item Same cleaned and imputed source.
		\item Unscaled continuous columns.
		\item Key and meter remain categorical.
		\item LightGBM histogram splits.
	\end{itemize}
	\end{block}
\end{columns}

\medskip

Normalization helps when input scales are not commensurate.~{\scriptsize\autocite{guyon2003selection}} PCA is learned only from the linear train view.~{\scriptsize\autocite{jolliffe2016pca}} LightGBM supplies the histogram-based tree comparison.~{\scriptsize\autocite{ke2017lightgbm}}

\medskip

\begin{block}{Shared contract}
Both views freeze the same predictors, missing-value rules, transform order, feature order, and fitted statistics. IDs remain for audit only.
\end{block}

\end{frame}

\begin{frame}{Key encoding is a controlled contract choice}

\begin{center}
\setlength{\tabcolsep}{4pt}
\begin{tabular}{l p{.34\textwidth} p{.45\textwidth}}
	\textbf{Contract} & \textbf{Encoding} & \textbf{Meaning} \\
	\hline
	K0 & Omit key & Does key help? \\
	K1 & One-hot + unknown & No imposed geometry; default \\
	K2 & Chromatic sine/cosine & Semitone-circle geometry \\
	K3 & Fifths-circle sine/cosine & Tonal fifths geometry \\
\end{tabular}
\end{center}

\medskip
\medskip

The circle of fifths makes K3 plausible for musical key, but does not prove it predicts year better.~{\scriptsize\autocite{kania2019key}}

\medskip
\medskip

\begin{block}{Controlled change}
Only the key representation changes; rows, all other predictors, and the Ridge baseline stay fixed.
\end{block}

\end{frame}

\begin{frame}{Result: key encoding has negligible impact}

\begin{center}
{
\setlength{\tabcolsep}{4pt}
\begin{tabular}{l r r r}
	\textbf{Contract} & \textbf{Dim} & \textbf{Validation MAE} & \textbf{Validation RMSE} \\
	\hline
	K0: no key & $110$ & $6.4758$ & $8.9534$ \\
	K1: one-hot & $122$ & $6.4750$ & $8.9513$ \\
	K2: chromatic circle & $112$ & $6.4757$ & $8.9533$ \\
	K3: fifths circle & $112$ & $6.4756$ & $8.9526$ \\
\end{tabular}
}
\end{center}

{\small
All four contracts use the same $420{,}013$ training tracks, $46{,}127$ validation tracks, Ridge settings, and 100 optimization iterations. The entire MAE spread is below $0.001$ years.
}

\medskip
\medskip

\begin{block}{Selected linear contract: K0}
Key encoding provides no practically meaningful improvement, while adding dimensions to the model. We therefore omit key from the linear view; the raw categorical key remains available for a separate tree-model ablation.
\end{block}

\end{frame}
